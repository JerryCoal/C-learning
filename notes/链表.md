# 链表知识


## 1.链表简介：
    （1）链表不是线性存放的，而是靠指针前后串接形成的
    （2）包括数据域和指针域
    （3）单链表，双链表，循环链表

## 2.C语言链表定义实例:
```c

struct Node {
    int a;
    int b;
    //...........
    struct Node *next;
};
```

```c
//反链表读取
// 参数：head 是原链表的头节点
// 返回值：反转后的新链表头节点
struct Node* reverse_list(struct Node* head) {
    struct Node* new_head = NULL; // 新链表的头节点
    struct Node* p = head;        // 遍历指针，指向当前节点

    while (p != NULL) {
        struct Node* next = p->link; // 【关键】先保存下一个节点！
        
        // 1. 把当前节点 p 插到新链表的头部
        p->link = new_head;
        // 2. 更新新链表的头节点为 p
        new_head = p;

        p = next; // 3. 指针后移，处理下一个节点
    }

    return new_head; // 返回反转后的头
}

// 获取第 pos 个位置的元素，pos 从 1 开始
int getByPos(Node *head, int pos) {
    Node *p = head;

    // 向后走 pos-1 步
    for (int i = 1; i < pos; i++) {
        if (p == NULL) break;  // 防止越界
        p = p->next;
    }

    if (p == NULL) {
        printf("位置非法\n");
        return -1; // 错误标记
    }

    return p->data;
}

```

## 4.C语言链表的插入与删除元素
```c

// 把 target 插入到 dest 链表中，保持 大的在前，小的在后
void insert_item(struct Node* target, struct Node* dest) {
    struct Node* p = dest;

    // 从头找【应该插入的位置】
    while (p->link != NULL && p->link->data > target->data) {
        p = p->link;
    }

    //  标准插入步骤（必须这两句）
    target->link = p->link;
    p->link = target;
}



//链表删除
void delete_node(struct node** head,struct node* target){
    // 若是空指针
    if (*head == NULL || target == NULL){
        return ;
    }
    // 2. 要删除的是 头节点
    if (*head == target) {
        *head = target->link;  // 头指针后移
        free(target);
        return;
    }
    // 找 target 的 前驱节点
    struct node *prev = *head;
    while (prev->link != NULL && prev->link != target) {
        prev = prev->link;
    }

    // 没找到目标节点
    if (prev->link == NULL)
        return;

    // 4. 删除节点
    prev->link = target->link;
    free(target);
}
```


## 5.循环链表的创建
```c
// 假设你的结构体是这样（必须有 link）
struct node {
    int place;
    int data;
    struct node *link;
};

// --------------- 正确创建循环链表 ---------------
struct node *create_circular_list(int n) {
    struct node *list = NULL;  // 头指针
    struct node *tail = NULL;  // 尾指针
    struct node *p = NULL;
    int x, y;

    for (int i = 0; i < n; i++) {
        // 申请新节点
        p = (struct node *)malloc(sizeof(struct node));
        scanf("%d %d", &x, &y);
        p->place = x;
        p->data = y;

        // 第一个节点：初始化头尾
        if (list == NULL) {
            list = p;
            tail = p;
        }
        // 不是第一个节点：追加到尾部
        else {
            tail->link = p;
            tail = p;
        }
    }

    // 【关键】最后让尾部指向头部，形成循环
    if (tail != NULL)
        tail->link = list;

    return list;
}

//循环链表最关键的在于循环遍历的时候要检测p != head (!!!!!!!)
//推荐使用do{....}while();

```


## 总结：链表操作注意：
    1.必须考虑链表为空的安全问题
    2.链表初始化后再读入数据
    3.分清楚循环链表还是单链表
    4.函数中传入参数时使用头指针的地址可以修改整个链表的数据
